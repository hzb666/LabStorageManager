### 2.4 文件上传安全

#### 问题 2.4.1: 图片上传缺少文件内容验证 [中危]

**文件**: `app/services/image_service.py:85-129`

**问题**: 只验证 `content_type`，攻击者可以伪造 MIME 类型上传恶意文件

**修复建议**: 添加文件魔数验证
```python
ALLOWED_IMAGE_MAGICS = {
    "jpeg": b"\xFF\xD8\xFF",
    "png": b"\x89\x50\x4E\x47",
    "webp": b"RIFF"
}

def validate_image_magic(file: UploadFile) -> bool:
    header = file.file.read(16)
    file.file.seek(0)
    for magic in ALLOWED_IMAGE_MAGICS.values():
        if header.startswith(magic):
            return True
    return False
```

      reagentOrderAPI.getMyOrders(),
      consumableOrderAPI.getMyOrders(),

export function useDebounce<T>(value: T, delay: number): T {
  const [debouncedValue, setDebouncedValue] = useState<T>(value)
  useEffect(() => {
    const handler = setTimeout(() => setDebouncedValue(value), delay)
    return () => clearTimeout(handler)
  }, [value, delay])
  return debouncedValue
}

// 2. 建议添加: useToggle
export function useToggle(initialValue = false): [boolean, () => void] {

  return [value, toggle]
}
